void	child_for_one_cmd(t_execution *cmd, t_env **env)
{
	char	*path;

	path = find_path(cmd, env);
	if (!path)
	{
		if (access(cmd->cmd[0], X_OK) == -1 && !get_env_value("PATH", *env))
		{
			(ft_putstr_fd("minishell: ", 2), ft_putstr_fd(cmd->cmd[0], 2));
			(perror(" "), exit(126));
		}
		else if (access(cmd->cmd[0], X_OK) == 0 && !get_env_value("PATH", *env))
			exit (0);
		ft_putstr_fd(cmd->cmd[0], 2);
		ft_putstr_fd(": command not found\n", 2);
		exit(127);
	}
	cmd->path = path;
	cmd->env = change_envlist_to_envchar(*env);
	signal_handler(0, *env);
	printf("thiiiiiis \n");
	if (execve(cmd->path, cmd->cmd, cmd->env) == -1)
	{
		printf("hola\n ");
		printf("cmd.flag = %d\n", cmd->flag_expand);
		printf("cmd.cmd[0] = %s\n", cmd->cmd[0]);
		if (!ft_strcmp(cmd->cmd[0], "..") || !cmd->flag_expand)
			(print_err("command not found"), exit(127));
		perror("minisehll");
	}
}
